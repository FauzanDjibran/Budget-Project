table fin_cash_bank_transaction {
  id                            bigint [pk, increment, not null]

  transaction_no                varchar(255) [not null, unique]
  document_date                 date [not null]
  posting_date                  datetime

  transaction_type              enum('In', 'Out') [not null] // sesuai dengan menu UI IN, OUT, Transfer

  company_id                    bigint [not null, ref : > sys_company.id]

  purpose                       varchar(255) [not null] // nanti akan berkembang sesuai kebutuhan
  //OUT :
  // 

  //IN : 
  // 

  cash_bank_id                  bigint [ref : >? m_cash_bank.id]

  currency_id                   bigint [not null, ref : > ref_currency.id] // currency dari bank
  exchange_rate                 decimal(18,6) [not null, default: 1]

  partner_id                    bigint [ref : >? m_partner.id] // tidak semua purpose punya partner

  transaction_amount            decimal(18,2) [not null]
  transaction_base_amount       decimal(18,2) [not null] // akumulasi base dari line 

  note                          text

  status                        enum('Draft', 'Posted', 'Cancelled') [not null, default: 'Draft']

  created_by                    bigint [not null]
  updated_by                    bigint

  created_at                    datetime [not null, default: `CURRENT_TIMESTAMP`]
  updated_at                    datetime [not null, default: `CURRENT_TIMESTAMP`]                
}

table fin_cash_bank_transaction_line {
  id                            bigint [pk, increment, not null]

  transaction_id                bigint [not null, ref : > fin_cash_bank_transaction.id]
  sequence_no                   int [not null]

  // reference bebas agar bisa menampung berbagai macam dokumen
  source_doc_type_id            bigint [not null, ref : > sys_doc_type.id]
  source_doc_id                 bigint [not null]

  outstanding_amount            decimal(18,2) [not null] // snapshot brp yang bisa dibayar pada saat itu, selalu dalam mata uang asli
  // cukup nilai asli, base outstanding kurang bermanfaat

  settlement_amount             decimal(18,2) [not null] // nominal source dokumen yang diselesaikan dalam mata uang asli
  settlement_base_amount        decimal(18,2) [not null] // nominal source dokumen yang diselesaikan dalam mata uang base
  
  settlement_exchange_rate      decimal(18,6)
  // conversion rate dari source document currency
  // ke cash/bank transaction currency
  // hanya digunakan jika conversion diperlukan

  transaction_amount            decimal(18,2) [not null] // // nominal cash/bank yang bergerak dalam currency header
  transaction_base_amount       decimal(18,2) [not null] // nominal cash / bank yang bergerak dalam base (menggunakan catatan)

  created_by                    bigint [not null]
  updated_by                    bigint

  created_at                    datetime [not null, default: `CURRENT_TIMESTAMP`]
  updated_at                    datetime [not null, default: `CURRENT_TIMESTAMP`]                
}

table fin_funding_request {
  id                            bigint [pk, increment, not null]

  funding_request_no            varchar(255) [not null, unique]

  request_date                  date [not null]

  // Realization / Cash Bank Transaction yang membutuhkan treasury
  transaction_id                bigint [not null, ref: > fin_cash_bank_transaction.id]

  // Company yang meminta / membutuhkan treasury
  requesting_company_id         bigint [not null, ref: > sys_company.id]

  // Company yang menjadi Treasury Provider
  treasury_provider_company_id  bigint [not null, ref: > sys_company.id]

  // IN  = Provider menerima dari Requesting Company
  // OUT = Provider memberi kepada Requesting Company
  direction                     enum('In', 'Out') [not null]

  // Currency kebutuhan / settlement
  currency_id                   bigint [not null, ref: > ref_currency.id]

  // Nominal kebutuhan dalam currency di atas.
  // Tidak memiliki exchange rate dan base amount.
  request_amount                decimal(18,6) [not null]

  // Bank yang dipilih oleh Treasury Provider.
  // NULL ketika Request masih Open.
  provider_cash_bank_id         bigint [ref: >? m_cash_bank.id]

  status                        enum('Open', 'Closed') [not null, default: 'Open']

  note                          text

  created_by                    bigint [not null]
  updated_by                    bigint

  created_at                    datetime [not null, default: `CURRENT_TIMESTAMP`]
  updated_at                    datetime [not null, default: `CURRENT_TIMESTAMP`]
}
